Artifact and Purpose

Travlr Getaways is my CS 465 MongoDB, Express, Angular, and Node.js application. Its public repository records a completed project commit on August 24, 2026. The original artifact supports customer trip browsing and administrative trip management. I selected it for this enhancement because choosing a small set of relevant trips provides a concrete setting for comparing algorithms and explaining time and space tradeoffs. The Module Four version carries forward the verified Module Three controller, service, and repository separation.

Implemented Enhancement

The application now ranks trips by a budget and a preferred duration. Budget acts as a price ceiling. The score combines affordability at 60% and duration fit at 40%. These weights are explicit demonstration choices, not measured customer preferences. Records with malformed prices or durations are excluded. Scores are ordered from highest to lowest; ties use ascending trip code and then original input order. Rounding occurs only in the display. A new customer page calls a read-only API, explains the scoring rule, and displays matching trips or a clear empty result.

The full-sort baseline scores all eligible trips, sorts them, and returns the first k. The enhanced algorithm scans the same input while retaining at most k candidates in a bounded min-heap. Its root is the worst retained candidate. A better candidate replaces that root, after which the heap order is restored. A candidate no better than the root can be discarded because k better or equal candidates already exist. The retained subset is finally sorted for display. This invariant explains why the two algorithms should produce the same ordered result.

Complexity and Measured Results

For n input trips, m eligible trips, and r = min(k, m), full sorting takes O(n + m log m) time and O(m) auxiliary space. Heap selection takes O(n log(k + 1) + r log(r + 1)) time and O(k) auxiliary space for positive k. The final subset sort is included in that bound. A zero-k request returns immediately without a database call. These are ranking costs: the current repository still materializes all n database records, so the complete request has O(n) input storage. Query validation caps API results at 100 and limits the budget and duration before database access.

The benchmark used seeded synthetic inputs, three warmups, and seven measured repetitions with alternating execution order. Every benchmark result matched the baseline. With 100,000 inputs and k = 5, median sorting time was 41.300 milliseconds and heap time was 25.305 milliseconds. Sorting retained 74,855 scored candidates; the heap retained 5. With 1,000 inputs and k = 1,000, both retained 762 candidates and had similar measured runtime. This shows why a small requested subset matters. Retained counts measure the data structure, not total process memory. Machine load, runtime optimization, and garbage collection affect timing; the benchmark excludes database and browser time and was not run on an otherwise idle machine.

Verification and Reflection

The complete backend suite passed 31 tests with no failures or skips. It includes hand-calculated scores, deterministic ties, duplicate-code stability, invalid preferences, malformed legacy data, empty input, zero and oversized k, and 120 seeded randomized comparisons with the baseline. Seventeen additional checks passed against real temporary MongoDB 8.2.6, including recommendations over persisted records and rejection of an invalid query. The Angular production build passed. In Chrome, a budget of 1,500 and five preferred days returned Gale Reef followed by Dawson's Reef; a budget of 100 produced no matches. The screenshot and raw logs preserve those results.

The main technical challenge was making the heap comparison agree with the baseline at every boundary. A worst-first heap needs the opposite ordering from the final displayed list. Tie rules must also be applied during replacement, not only at the last sort. Another lesson is that a small auxiliary structure does not remove the cost of retrieving the input. Keeping the database boundary visible prevents an unsupported claim that the entire endpoint uses O(k) memory. Streaming and index work remain separate opportunities for Module Five.

const Trip = require('../models/travlr');

const tripFields = [
  'code',
  'name',
  'length',
  'start',
  'resort',
  'perPerson',
  'image',
  'description',
];

function tripPayload(body = {}) {
  return tripFields.reduce((payload, field) => {
    if (body[field] !== undefined) {
      payload[field] = body[field];
    }
    return payload;
  }, {});
}

function requiredFieldsMissing(payload) {
  return tripFields.filter((field) => (
    payload[field] === undefined
    || payload[field] === null
    || (typeof payload[field] === 'string' && payload[field].trim() === '')
  ));
}

function databaseError(res, error, action) {
  if (error?.name === 'ValidationError' || error?.name === 'CastError') {
    return res.status(400).json({ message: 'Invalid trip data', details: error.message });
  }

  if (error?.code === 11000) {
    return res.status(409).json({ message: 'A trip with that code already exists' });
  }

  return res.status(500).json({ message: `Unable to ${action} trip` });
}

function createTripsController(TripModel) {
  async function tripsList(req, res) {
    try {
      const trips = await TripModel.find({});

      if (trips.length === 0) {
        return res.status(404).json({ message: 'No trips found' });
      }

      return res.status(200).json(trips);
    } catch (error) {
      return res.status(500).json({ message: 'Unable to retrieve trips' });
    }
  }

  async function tripsFindByCode(req, res) {
    const tripCode = req.params?.tripCode?.trim();

    if (!tripCode) {
      return res.status(400).json({ message: 'Trip code is required' });
    }

    try {
      const trip = await TripModel.findOne({ code: tripCode });

      if (!trip) {
        return res.status(404).json({ message: `Trip ${tripCode} not found` });
      }

      return res.status(200).json(trip);
    } catch (error) {
      return res.status(500).json({ message: 'Unable to retrieve trip' });
    }
  }

  async function tripsAddTrip(req, res) {
    const payload = tripPayload(req.body);
    const missingFields = requiredFieldsMissing(payload);

    if (missingFields.length > 0) {
      return res.status(400).json({
        message: `Missing required trip fields: ${missingFields.join(', ')}`,
      });
    }

    try {
      const trip = await TripModel.create(payload);
      return res.status(201).json(trip);
    } catch (error) {
      return databaseError(res, error, 'create');
    }
  }

  async function tripsUpdateTrip(req, res) {
    const tripCode = req.params?.tripCode?.trim();

    if (!tripCode) {
      return res.status(400).json({ message: 'Trip code is required' });
    }

    const payload = tripPayload(req.body);
    const missingFields = requiredFieldsMissing(payload);

    if (missingFields.length > 0) {
      return res.status(400).json({
        message: `Missing required trip fields: ${missingFields.join(', ')}`,
      });
    }

    try {
      const trip = await TripModel.findOneAndUpdate(
        { code: tripCode },
        payload,
        { returnDocument: 'after', runValidators: true },
      );

      if (!trip) {
        return res.status(404).json({ message: `Trip ${tripCode} not found` });
      }

      return res.status(200).json(trip);
    } catch (error) {
      return databaseError(res, error, 'update');
    }
  }

  async function tripsDeleteTrip(req, res) {
    const tripCode = req.params?.tripCode?.trim();

    if (!tripCode) {
      return res.status(400).json({ message: 'Trip code is required' });
    }

    try {
      const trip = await TripModel.findOneAndDelete({ code: tripCode });

      if (!trip) {
        return res.status(404).json({ message: `Trip ${tripCode} not found` });
      }

      return res.status(200).json({ message: `Trip ${tripCode} deleted` });
    } catch (error) {
      return databaseError(res, error, 'delete');
    }
  }

  return {
    tripsList,
    tripsFindByCode,
    tripsAddTrip,
    tripsUpdateTrip,
    tripsDeleteTrip,
  };
}

module.exports = {
  ...createTripsController(Trip),
  createTripsController,
};
Enhanced app_api/algorithms/ranking.js
/** Chikeziri Ohaeri | CS 499 Module 4 | 2026-09-27 | AI-assisted.
 * Rank eligible trips by affordability (60%) and duration fit (40%).
 * Larger scores win; ties use ascending code, then original input order.
 * The heap keeps the WORST retained candidate at its root for replacement.
 * For n inputs, m eligible and r=min(k,m): sort O(n+m log m), O(m) space;
 * heap O(n log(k+1)+r log(r+1)), O(k) auxiliary space, excluding input.
 * Database list materialization still takes O(n) memory outside this module.
 */
const { ApiError } = require('../errors/api-error');
function preferences(query) {
  const bad = () => { throw new ApiError(400, 'INVALID_PREFERENCES', 'Use budget > 0 and <= 1000000, days 1-366, and k 0-100; no other parameters.'); };
  if (!query || Object.keys(query).some(key => !['budget','days','k'].includes(key))) bad();
  const raw = { budget: query.budget, days: query.days, k: query.k ?? '5' };
  for (const value of Object.values(raw)) if (typeof value !== 'string' || !/^\d+(?:\.\d{1,2})?$/.test(value)) bad();
  const result = Object.fromEntries(Object.entries(raw).map(([key,value])=>[key,Number(value)]));
  if (!(result.budget > 0 && result.budget <= 1e6) || !Number.isInteger(result.days) || result.days < 1 || result.days > 366 || !Number.isInteger(result.k) || result.k < 0 || result.k > 100) bad();
  return result;
}
function validOptions({budget,days,k}) {
  if (!Number.isFinite(budget) || budget <= 0 || !Number.isInteger(days) || days < 1 || !Number.isInteger(k) || k < 0) throw new RangeError('Invalid ranking options');
}
function candidate(trip,index,{budget,days}) {
  if (!trip || typeof trip.code !== 'string' || !/^[A-Za-z0-9_-]{1,40}$/.test(trip.code)) return null;
  const price = String(trip.perPerson ?? '');
  const duration = /^(\d+) nights?\s*\/\s*(\d+) days?$/.exec(String(trip.length ?? ''));
  if (!/^\d+(?:\.\d{1,2})?$/.test(price) || !duration) return null;
  const cost = Number(price), nights = Number(duration[1]), actualDays = Number(duration[2]);
  if (!Number.isFinite(cost) || cost > budget || actualDays < 1 || actualDays > 366 || nights !== actualDays-1) return null;
  const score = 0.6*(1-cost/budget)+0.4*(1-Math.abs(actualDays-days)/Math.max(actualDays,days));
  return {trip,score,index};
}
// Negative means a is better than b. No rounding before ordering.
function compare(a,b) { return b.score-a.score || (a.trip.code < b.trip.code ? -1 : a.trip.code > b.trip.code ? 1 : 0) || a.index-b.index; }
function result(items,stats) { return {results:items.map(({trip,score})=>({trip,score})),stats}; }
function rankSort(trips,options) {
  validOptions(options);
  const stats={examined:0,eligible:0,maxRetained:0,comparisons:0};
  if (!options.k) return result([],stats);
  const scored=[];
  for (const trip of trips) { const c=candidate(trip,stats.examined++,options); if(c) scored.push(c); }
  stats.eligible=stats.maxRetained=scored.length;
  scored.sort((a,b)=>{stats.comparisons++;return compare(a,b);});
  return result(scored.slice(0,options.k),stats);
}
function rankHeap(trips,options) {
  validOptions(options);
  const stats={examined:0,eligible:0,maxRetained:0,comparisons:0};
  if (!options.k) return result([],stats);
  const heap=[];
  const cmp=(a,b)=>{stats.comparisons++;return compare(a,b);};
  for (const trip of trips) {
    const c=candidate(trip,stats.examined++,options);
    if(!c) continue;
    stats.eligible++;
    if(heap.length < options.k) {
      heap.push(c); let i=heap.length-1;
      while(i>0) { const parent=(i-1)>>1; if(cmp(heap[i],heap[parent])<=0) break; [heap[i],heap[parent]]=[heap[parent],heap[i]]; i=parent; }
      stats.maxRetained=Math.max(stats.maxRetained,heap.length);
    } else if(cmp(c,heap[0])<0) {
      heap[0]=c; let i=0;
      while(true) {
        const left=2*i+1, right=left+1; let worst=i;
        if(left<heap.length && cmp(heap[left],heap[worst])>0) worst=left;
        if(right<heap.length && cmp(heap[right],heap[worst])>0) worst=right;
        if(worst===i) break;
        [heap[i],heap[worst]]=[heap[worst],heap[i]]; i=worst;
      }
    }
  }
  heap.sort(cmp); // The retained subset must also be returned in best-first order.
  return result(heap,stats);
}
module.exports={preferences,rankSort,rankHeap};
